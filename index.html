// ---------- LOGIN ---------- document.getElementById("login-form").addEventListener("submit", async (e)=>{ e.preventDefault(); clearMsg("login-msg"); const email = document.getElementById("login-email").value.trim(); const password = document.getElementById("login-password").value; const btn = e.target.querySelector("button"); btn.disabled = true; try{ const { data, error } = await supabase.auth.signInWithPassword({ email, password }); if(error){ showMsg("login-msg", error.message === "Invalid login credentials" ? "Email o password non corrette." : error.message, "error"); return; } currentUser = data.user; showFeed(); } catch(err){ showMsg("login-msg", "Impossibile contattare Supabase. Controlla di aver inserito URL e chiave corretti nel codice.", "error"); } finally { btn.disabled = false; } }); // ---------- PASSWORD RESET ---------- document.getElementById("reset-form").addEventListener("submit", async (e)=>{ e.preventDefault(); clearMsg("reset-msg"); const email = document.getElementById("reset-email").value.trim(); const btn = e.target.querySelector("button"); btn.disabled = true; try{ const { error } = await supabase.auth.resetPasswordForEmail(email); if(error){ showMsg("reset-msg", error.message, "error"); return; } showMsg("reset-msg", "Se l'indirizzo esiste, riceverai un'email con le istruzioni.", "success"); } catch(err){ showMsg("reset-msg", "Impossibile contattare Supabase. Controlla di aver inserito URL e chiave corretti nel codice.", "error"); } finally { btn.disabled = false; } }); // ---------- LOGOUT ---------- document.getElementById("logout-btn").addEventListener("click", async ()=>{ const btn = document.getElementById("logout-btn"); if(btn.disabled) return; btn.disabled = true; await supabase.auth.signOut(); currentUser = null; btn.disabled = false; showAuth(); }); // ---------- PUBLISH POST ---------- document.getElementById("publish-btn").addEventListener("click", async ()=>{ const btn = document.getElementById("publish-btn"); if(btn.disabled) return; const textarea = document.getElementById("post-content"); const content = textarea.value.trim(); if(!content || !currentUser) return; btn.disabled = true; const { error } = await supabase.from("posts").insert({ user_id: currentUser.id, content: content }); btn.disabled = false; if(error){ alert("Errore nella pubblicazione: " + error.message); return; } textarea.value = ""; loadPosts(); }); // ---------- LOAD POSTS (feed) ---------- async function loadPosts(){ const list = document.getElementById("posts-list"); list.innerHTML = '
Caricamento post...
'; const { data: posts, error } = await supabase .from("posts") .select("id, content, created_at, user_id, profiles(full_name), likes(user_id)") .order("created_at", { ascending: false }); if(error){ list.innerHTML = '
Errore nel caricamento: ' + escapeHtml(error.message) + '
'; return; } if(!posts || posts.length === 0){ list.innerHTML = '
Nessun post ancora. Scrivi il primo!
'; return; } list.innerHTML = posts.map(p=>{ const authorName = p.profiles?.full_name || "Utente"; const likeCount = p.likes ? p.likes.length : 0; const likedByMe = p.likes ? p.likes.some(l=>l.user_id === currentUser.id) : false; return `
${escapeHtml(initials(authorName))}
${escapeHtml(authorName)}
${timeAgo(p.created_at)}
${escapeHtml(p.content)}
`; }).join(""); document.querySelectorAll(".like-btn").forEach(btn=>{ btn.addEventListener("click", ()=>toggleLike(btn)); }); } // ---------- LIKE / UNLIKE ---------- async function toggleLike(btn){ if(btn.disabled) return; const postId = btn.dataset.postId; const liked = btn.dataset.liked === "true"; btn.disabled = true; if(liked){ await supabase.from("likes").delete() .eq("post_id", postId) .eq("user_id", currentUser.id); } else { await supabase.from("likes").insert({ post_id: postId, user_id: currentUser.id }); } btn.disabled = false; loadPosts(); } // ---------- SESSIONE ESISTENTE ---------- (async ()=>{ const { data } = await supabase.auth.getSession(); if(data.session){ currentUser = data.session.user; showFeed(); } })(); supabase.auth.onAuthStateChange((event, session)=>{ if(session) currentUser = session.user; });
